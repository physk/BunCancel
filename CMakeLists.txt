cmake_minimum_required(VERSION 3.20)
project(BunCancel VERSION 0.1.3 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(buncancel-dsp STATIC src/dsp/aec.cpp)
target_include_directories(buncancel-dsp PUBLIC src/dsp)
set_target_properties(buncancel-dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_executable(buncancel-offline tools/offline.cpp)
target_link_libraries(buncancel-offline PRIVATE buncancel-dsp)

include(CTest)
if(BUILD_TESTING)
    add_executable(buncancel-dsp-tests tests/dsp/test_aec.cpp)
    target_link_libraries(buncancel-dsp-tests PRIVATE buncancel-dsp)
    add_test(NAME dsp COMMAND buncancel-dsp-tests)
    add_executable(buncancel-allocation-tests tests/dsp/test_allocations.cpp)
    target_link_libraries(buncancel-allocation-tests PRIVATE buncancel-dsp)
    add_test(NAME dsp-allocations COMMAND buncancel-allocation-tests)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    add_test(NAME offline-cli COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/dsp/test_offline.py $<TARGET_FILE:buncancel-offline>)
endif()

option(BUNCANCEL_BUILD_VST3 "Build the experimental Windows x64 VST3" OFF)
if(BUILD_TESTING OR BUNCANCEL_BUILD_VST3)
    add_subdirectory(src/capture/windows)
endif()
if(BUILD_TESTING)
    add_executable(buncancel-timeline-tests tests/capture/test_timeline.cpp)
    target_link_libraries(buncancel-timeline-tests PRIVATE buncancel-capture)
    add_test(NAME reference-timeline COMMAND buncancel-timeline-tests)

    add_executable(buncancel-pipeline-tests tests/integration/test_reference_pipeline.cpp)
    target_include_directories(buncancel-pipeline-tests PRIVATE src src/plugin)
    target_link_libraries(buncancel-pipeline-tests PRIVATE buncancel-capture buncancel-dsp)
    add_test(NAME reference-pipeline COMMAND buncancel-pipeline-tests)

    add_executable(buncancel-post-alignment-tests tests/integration/test_post_alignment.cpp)
    target_include_directories(buncancel-post-alignment-tests PRIVATE src src/plugin)
    target_link_libraries(buncancel-post-alignment-tests PRIVATE buncancel-capture buncancel-dsp)
    add_test(NAME post-alignment COMMAND buncancel-post-alignment-tests)
endif()
if(BUNCANCEL_BUILD_VST3)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The reference capture plugin requires 64-bit Windows")
    endif()
    add_subdirectory(src/plugin)
endif()
